const matchCreatorNames = (nameA?: string, nameB?: string): boolean => {
  if (!nameA || !nameB) return false;
  const cleanA = String(nameA).trim().toLowerCase().replace(/\s+/g, ' ');
  const cleanB = String(nameB).trim().toLowerCase().replace(/\s+/g, ' ');
  if (!cleanA || !cleanB) return false;
  if (cleanA === cleanB || cleanA.includes(cleanB) || cleanB.includes(cleanA)) return true;
  for (const group of CREATOR_ALIAS_GROUPS) {
    const aIn = group.some(a => cleanA === a || cleanA.includes(a) || a.includes(cleanA));
    const bIn = group.some(b => cleanB === b || cleanB.includes(b) || b.includes(cleanB));
    if (aIn && bIn) return true;
  }
  return false;
};

const loadCorsImage = (url: string): Promise<HTMLImageElement | null> => {
  return new Promise(resolve => {
    if (!url) return resolve(null);
    const img = new Image();
    img.crossOrigin = 'anonymous';
    img.onload = () => resolve(img);
    img.onerror = () => resolve(null);
    img.src = url;
  });
};

const drawRoundedRect = (
  ctx: CanvasRenderingContext2D,
  x: number,
  y: number,
  w: number,
  h: number,
  r: number
) => {
  const radius = Math.min(r, w / 2, h / 2);
  ctx.beginPath();
  ctx.moveTo(x + radius, y);
  ctx.arcTo(x + w, y, x + w, y + h, radius);
  ctx.arcTo(x + w, y + h, x, y + h, radius);
  ctx.arcTo(x, y + h, x, y, radius);
  ctx.arcTo(x, y, x + w, y, radius);
  ctx.closePath();
};

      ctx.fill();
      ctx.font = '900 24px sans-serif';
      ctx.fillStyle = '#000000';
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      ctx.fillText((studioCtaLabel || 'UNLOCK NOW').toUpperCase(), W / 2, ctaY + ctaH / 2 + 2);

      const subY = ctaY - 26;
      ctx.font = '900 20px sans-serif';
      ctx.fillStyle = '#fe9a00';
      ctx.fillText(studioSubheadline.toUpperCase(), W / 2, subY);

      ctx.fillStyle = '#ffffff';
      const mFontSize = 52;
      ctx.font = `900 italic ${mFontSize}px sans-serif`;
      const maxMobileTextW = W - padX * 2;
      const words = studioHeadline.split(' ');
      let lines: string[] = [];
      let currentLine = '';
      words.forEach(w => {
        const test = currentLine ? `${currentLine} ${w}` : w;
        if (ctx.measureText(test).width > maxMobileTextW && currentLine) {
          lines.push(currentLine);
          currentLine = w;
        } else {
          currentLine = test;
        }
      });
      if (currentLine) lines.push(currentLine);
      lines = lines.slice(0, 2);

      let headlineTopY = subY - 38;
      ctx.shadowColor = 'rgba(0,0,0,0.95)';
      ctx.shadowBlur = 16;
      for (let i = lines.length - 1; i >= 0; i--) {
        ctx.fillText(lines[i], W / 2, headlineTopY);
        headlineTopY -= mFontSize + 6;
      }
      ctx.shadowBlur = 0;

      if (loadedAvatars.length > 0) {

          ctx.arc(cx, avCenterY, avSize / 2, 0, Math.PI * 2);
          ctx.lineWidth = 4;
          ctx.strokeStyle = '#fe9a00';
          ctx.stroke();
        });
      }
      ctx.textAlign = 'left';

          canvas.width = 0;

// Generates an ISO timestamp corresponding to 9:00 AM Eastern Time (America/New_York) for a given YYYY-MM-DD
export const toNineAmEstIso = (dateStr: string, timeStr: string = '09:00'): string => {
  const cleanTime = (timeStr || '09:00').slice(0, 5);
  const estDate = new Date(`${dateStr}T${cleanTime}:00-05:00`);
  if (!isNaN(estDate.getTime())) {
    return estDate.toISOString();
  }
  const [year, month, day] = dateStr.split('-').map(Number);
  const [hour, minute] = cleanTime.split(':').map(Number);
  return new Date(year, month - 1, day, hour, minute, 0).toISOString();
};

// Formats a Date or ISO timestamp into YYYY-MM-DD in Eastern Time so UTC offsets never shift the day
export const formatDateInput = (d: Date): string => {
  try {
    const parts = new Intl.DateTimeFormat('en-CA', {
      timeZone: 'America/New_York',
      year: 'numeric',
      month: '2-digit',
      day: '2-digit'
    }).formatToParts(d);
    const y = parts.find(p => p.type === 'year')?.value;
    const m = parts.find(p => p.type === 'month')?.value;
    const day = parts.find(p => p.type === 'day')?.value;
    if (y && m && day) return `${y}-${m}-${day}`;
  } catch {
    // Fallback to local calendar components
  }
  const year = d.getFullYear();
  const month = String(d.getMonth() + 1).padStart(2, '0');
  const day = String(d.getDate()).padStart(2, '0');
  return `${year}-${month}-${day}`;
};

const getEstDayOfWeek = (iso: string): number => {
  try {
    const shortDay = new Intl.DateTimeFormat('en-US', {
      timeZone: 'America/New_York',
      weekday: 'short'
    }).format(new Date(iso));
    const map: Record<string, number> = {
      Sun: 0,
      Mon: 1,
      Tue: 2,
      Wed: 3,
      Thu: 4,
      Fri: 5,
      Sat: 6
    };
    if (map[shortDay] !== undefined) return map[shortDay];
  } catch {
    // fallback
  }
  return new Date(iso).getDay();
};

export const doesWeekMatchCadence = (
  weekNum: number,
  cadence: CadencePattern,
  startWeekOffset: number
): boolean => {
  if (weekNum < startWeekOffset) return false;

  switch (cadence) {
    case 'weekly':
      return true;
    case 'odd_weeks':
      return weekNum % 2 === 1;
    case 'even_weeks':
      return weekNum % 2 === 0;
    case 'launch_then_even':
      return weekNum === 1 || weekNum % 2 === 0;
    case 'launch_then_odd':
      return weekNum % 2 === 1;
    case 'first_3_then_monthly':
      return weekNum <= 3 || (weekNum > 3 && (weekNum - 3) % 4 === 0);
    case 'monthly_w1':
      return (weekNum - 1) % 4 === 0;
    case 'monthly_w4':
      return weekNum % 4 === 0;
    default:
      return true;
  }
};

export const getUpcomingDatesForDay = (
  dayType: TrackDayType,
  totalWeeks: number = 104
): { weekNum: number; dateStr: string; label: string }[] => {
  const targetDow = dayType === 'MONDAY' ? 1 : dayType === 'WEDNESDAY' ? 3 : 6;
  const now = new Date();
  const cursor = new Date();
  cursor.setHours(0, 0, 0, 0);

  if (cursor.getDay() === targetDow && now.getHours() >= 9) {
    cursor.setDate(cursor.getDate() + 1);
  }

  while (cursor.getDay() !== targetDow) {
    cursor.setDate(cursor.getDate() + 1);
  }

  const results: { weekNum: number; dateStr: string; label: string }[] = [];
  for (let w = 1; w <= totalWeeks; w++) {
    const d = new Date(cursor);
    d.setDate(cursor.getDate() + (w - 1) * 7);
    const yyyy = d.getFullYear();
    const mm = String(d.getMonth() + 1).padStart(2, '0');
    const dd = String(d.getDate()).padStart(2, '0');
    results.push({
      weekNum: w,
      dateStr: `${yyyy}-${mm}-${dd}`,
      label: d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
    });
  }
  return results;
};

export const getMatchingDropDatesForTrack = (
  dayType: TrackDayType,
  cadence: CadencePattern,
  startWeekOffset: number,
  requiredDrops: number
): { weekNum: number; dateStr: string; label: string }[] => {
  const needed = Math.max(8, requiredDrops + 4);
  const weeksToGenerate = Math.max(52, needed * 5 + startWeekOffset);
  const allDates = getUpcomingDatesForDay(dayType, weeksToGenerate);
  return allDates.filter(d => doesWeekMatchCadence(d.weekNum, cadence, startWeekOffset));
};

            const sorted = [...unrelChaps].sort(
              (a, b) => Number(a.chapter_number) - Number(b.chapter_number)
            );
            const firstWithDate = sorted.find(c => Boolean(c.publish_at));
            let dbDayType: TrackDayType | null = null;

            if (firstWithDate?.publish_at) {
              const dow = getEstDayOfWeek(firstWithDate.publish_at);
              if (dow === 1) dbDayType = 'MONDAY';
              else if (dow === 3) dbDayType = 'WEDNESDAY';
              else if (dow === 6) dbDayType = 'SATURDAY';
            }

            const existing = bySlug.get(slug);
            const canonical = CANONICAL_SERIES_TRACK_DEFAULTS[slug.toLowerCase()];

            if (existing) {
              if (dbDayType && existing.dayType !== dbDayType) {
                bySlug.set(slug, {
                  ...existing,
                  dayType: dbDayType,
                  cadence: detectCadenceFromDbChapters(sorted, slug)
                });
              }
            } else {
              const resolvedDay = dbDayType || canonical?.dayType || 'SATURDAY';
              const resolvedCadence = detectCadenceFromDbChapters(sorted, slug);
              bySlug.set(slug, {
                id: `${resolvedDay}_${slug}`,
                seriesSlug: slug,
                dayType: resolvedDay,
                cadence: resolvedCadence,
                chaptersPerDrop: canonical?.chaptersPerDrop || 1,
                startWeekOffset: 1,
                paused: false
              });
            }
          });

          return Array.from(bySlug.values());
        });
      }

      setAllChapters(chaptersList);
    } catch (err: any) {
      console.error('Error loading chapters for scheduler:', err.message);
    } finally {
      setLoading(false);
    }
  }, [seriesList]);

  const handleUpdateTrackItem = async (id: string, patch: Partial<TrackAssignment>) => {
    const existing = trackAssignments.find(t => t.id === id);
    if (!existing) return;

    setTrackAssignments(prev => prev.map(item => (item.id === id ? updatedItem : item)));

    // Only touch DB if user changed Day, Cadence, Ch/Drop, or Start Week on an unpaused series with queued chapters
    const scheduleRuleChanged =
      patch.dayType !== undefined ||
      patch.cadence !== undefined ||
      patch.chaptersPerDrop !== undefined ||
      patch.startWeekOffset !== undefined;

    const seriesChaps = chaptersBySeries[updatedItem.seriesSlug] || [];
    const matchedSeries = seriesList.find((s: any) => s.slug === updatedItem.seriesSlug);

    if (scheduleRuleChanged && !updatedItem.paused && seriesChaps.some(isChapterUnreleased)) {
      if (
        !window.confirm(
          `Update queued chapter dates for "${matchedSeries?.title || updatedItem.seriesSlug}" to match this new rule? (Other series will NOT be affected.)`
        )
      ) {
        setTrackAssignments(prev => prev.map(item => (item.id === id ? existing : item)));
        return;
      }

      setSaving(true);
      try {
        const count = await syncSingleTrackToDatabase(updatedItem, seriesChaps);
        window.dispatchEvent(new Event('seriesUpdated'));
        window.dispatchEvent(new Event('vaultUpdated'));
        await fetchAllChapters();

        .from('chapters')
        .update({
          is_published: true,
          publish_at: publishIso
        })
        .eq('series_slug', seriesObj.slug);

    if (!selectedSeriesSlug || currentSeriesChapters.length === 0) return;
    const matchedSeries = seriesList.find((s: any) => s.slug === selectedSeriesSlug);
    const input = window.prompt(
      `Set Live vs. Scheduled Cutoff for ${matchedSeries?.title || selectedSeriesSlug}:\n\nEnter the HIGHEST Chapter Number that should be LIVE right now:`,
      '1'
    );
    if (!input) return;
    const cutoffNum = Number(input.trim());
    if (isNaN(cutoffNum) || cutoffNum < 0) return alert('Invalid chapter number.');

    setSaving(true);
    try {
      const toMakeLive = currentSeriesChapters.filter(
        (c: any) => Number(c.chapter_number) <= cutoffNum
      );
      const toSchedule = currentSeriesChapters.filter(
        (c: any) => Number(c.chapter_number) > cutoffNum
      );

      for (const ch of toMakeLive) {
        const isNewest = Number(ch.chapter_number) === cutoffNum;
        const ts = isNewest
          ? new Date(Date.now() - 60 * 60 * 1000).toISOString()
          : new Date(Date.now() - 14 * 24 * 60 * 60 * 1000).toISOString();
        const { error } = await supabase
          .from('chapters')
          .update({ is_published: true, publish_at: ts })
          .eq('id', ch.id);
        if (error) throw error;

      if (toSchedule.length > 0) {
        const assignedTrack = trackAssignments.find(t => t.seriesSlug === selectedSeriesSlug);
        const dayType = assignedTrack?.dayType || 'SATURDAY';
        const cadence = assignedTrack?.cadence || 'weekly';
        const countPerDrop = Math.max(1, assignedTrack?.chaptersPerDrop || 1);
        const validDates = getMatchingDropDatesForTrack(
          dayType,
          cadence,
          1,
          Math.ceil(toSchedule.length / countPerDrop) + 2
        );

        let cursor = 0;
        for (const dropDay of validDates) {
          if (cursor >= toSchedule.length) break;
          const publishIso = toNineAmEstIso(dropDay.dateStr, '09:00');
          for (let i = 0; i < countPerDrop; i++) {
            if (cursor >= toSchedule.length) break;
            const ch = toSchedule[cursor++];
            const { error } = await supabase
              .from('chapters')
              .update({ is_published: true, publish_at: publishIso })
              .eq('id', ch.id);
            if (error) throw error;
          }
        }
      }

      window.dispatchEvent(new Event('vaultUpdated'));
      window.dispatchEvent(new Event('seriesUpdated'));
      await fetchAllChapters();
      showNotice(
        `Updated ${matchedSeries?.title}: Ch. 1–${cutoffNum} Live, Ch. ${cutoffNum + 1}+ Scheduled.`
      );
    } catch (err: any) {
      alert('Error repairing cutoff: ' + err.message);
    } finally {
      setSaving(false);
    }

export const unlockCreatorAutographMaster = async (
  userId: string,
  creatorNameRaw?: string,
  eventLabel = 'Themed Annual Bundle'
): Promise<string | null> => {
  const targetCreator = String(creatorNameRaw || '').trim();
  if (!targetCreator) return null;

  try {
    const { data: existingLogs } = await supabase
      .from('bingo_signatures_log')
      .select('creator_name')
      .eq('user_id', userId);

    const alreadyHas = (existingLogs || []).some((row: any) =>
      doCreatorNamesMatch(row.creator_name, targetCreator)
    );

    if (!alreadyHas) {
      await supabase.from('bingo_signatures_log').insert([
        {
          user_id: userId,
          creator_name: targetCreator,
          event_name: eventLabel,
          signature_type: 'Standard'
        }
      ]);
    }

    const { data: allSigs } = await supabase
      .from('creator_signatures')
      .select('creator_name, signature_url');

    const matchedSig = (allSigs || []).find((r: any) =>
      doCreatorNamesMatch(r.creator_name, targetCreator)
    );

    const canonicalName = matchedSig?.creator_name || targetCreator;

    const localHunts: string[] = JSON.parse(localStorage.getItem('am_bingo_hunts') || '[]');
    if (!localHunts.some(h => doCreatorNamesMatch(h, targetCreator))) {
      localHunts.push(targetCreator);
    }
    if (canonicalName && !localHunts.includes(canonicalName)) {
      localHunts.push(canonicalName);
    }
    localStorage.setItem('am_bingo_hunts', JSON.stringify(localHunts));

    if (matchedSig?.signature_url) {
      const localSigs: Record<string, string> = JSON.parse(
        localStorage.getItem('am_bingo_sigs') || '{}'
      );
      localSigs[targetCreator] = matchedSig.signature_url;
      localSigs[canonicalName] = matchedSig.signature_url;
      localStorage.setItem('am_bingo_sigs', JSON.stringify(localSigs));
    }

    window.dispatchEvent(new Event('bingoUpdated'));
    return canonicalName;
  } catch (err) {
    console.error('Error unlocking autograph:', err);
    return targetCreator;
  }
};

export const grantMasterBundleRewardsToUser = async (
  userId: string,
  bundle: BundlePayload,
  alreadyPremium: boolean,
  eventLabel = 'Themed Annual Bundle'
): Promise<{ unlockedParts: string[]; unlockedCreators: string[]; activatedPlus: boolean }> => {
  const { data: currentProf } = await supabase
    .from('profiles')
    .select('*')
    .eq('id', userId)

  const now = new Date();
  const includeSub = bundle.include_1yr_sub !== false;
  const baseDate =
    currentProf?.premium_until && new Date(currentProf.premium_until) > now

    0,
    Number(currentProf?.hypes_remaining ?? currentProf?.super_hypes_left ?? 1)
  );
  const nextActiveHypes = includeSub
    ? Math.min(10, Math.max(3, alreadyPremium ? currentActiveHypes + 3 : currentActiveHypes + 2))
    : currentActiveHypes;

  const currentVaultedHypes = Math.max(0, Number(currentProf?.vaulted_hypes || 0));
  const extraVaultedHypes = Math.max(0, Number(bundle.bonus_hypes || 0));
  const nextVaultedHypes = currentVaultedHypes + extraVaultedHypes;

  const unlockedParts: string[] = [];
  const unlockedCreators: string[] = [];
  let bonusFandomPts = includeSub ? (alreadyPremium ? 30 : 15) : 0;

  const profilePatch: Record<string, any> = {
    hypes_remaining: nextActiveHypes,
    super_hypes_left: nextActiveHypes,
    vaulted_hypes: nextVaultedHypes
  };

  if (includeSub) {
    profilePatch.is_premium = true;
    profilePatch.premium_until = nextYearDate;
    unlockedParts.push(alreadyPremium ? '+1 Year PLUS Extended' : '1-Year PLUS Pass');
  }

  if (extraVaultedHypes > 0) {
    unlockedParts.push(`+${extraVaultedHypes} Vaulted Gold Hypes`);
  }

  let resolvedSkin1Url = bundle.card_skin_url || '';
  let resolvedSkin1Name = bundle.card_skin_name || '';
  if (bundle.card_skin_id && !resolvedSkin1Url) {
    const { data: skinRow } = await supabase
      .from('card_skins')
      .select('name, image_url')
      .eq('id', bundle.card_skin_id)
      .maybeSingle();
    if (skinRow) {

      resolvedSkin1Name = resolvedSkin1Name || skinRow.name || '';
    }
  }

    bundle.card_skin_id ? String(bundle.card_skin_id) : '',
    resolvedSkin1Url ? String(resolvedSkin1Url) : '',
    bundle.card_skin_2_id ? String(bundle.card_skin_2_id) : '',
    bundle.card_skin_2_url ? String(bundle.card_skin_2_url) : ''
  ].filter(Boolean);

  if (skinItemsToUnlock.length > 0) {
    const existingSkins: string[] = Array.isArray(currentProf?.unlocked_skins)
      ? currentProf.unlocked_skins.map(String)
      : [];
    const mergedSkins = Array.from(new Set([...existingSkins, ...skinItemsToUnlock]));
    profilePatch.unlocked_skins = mergedSkins;

    if (resolvedSkin1Url) {
      profilePatch.card_skin = resolvedSkin1Url;
    }
    if (bundle.card_skin_id) {
      profilePatch.card_skin_id = bundle.card_skin_id;
    }

    try {
      const localSkins: string[] = JSON.parse(localStorage.getItem('am_unlocked_skins') || '[]');

        JSON.stringify(Array.from(new Set([...localSkins, ...skinItemsToUnlock])))
      );
    } catch {
      // ignore
    }

      skinNames.length > 0 ? `${skinNames.join(' & ')} Card Skin(s)` : 'Exclusive Hype Card Skin(s)'
    );

  }

  const validAvatars = (bundle.avatar_urls || []).filter(Boolean);

      : [];
    const mergedAvatars = Array.from(new Set([...existingAvatars, ...validAvatars]));

    profilePatch.avatar_url = validAvatars[0];

    try {
      const localAvs: string[] = JSON.parse(localStorage.getItem('am_unlocked_avatars') || '[]');
      const mergedLocal = Array.from(new Set([...localAvs, ...validAvatars]));
      localStorage.setItem('am_unlocked_avatars', JSON.stringify(mergedLocal));
    } catch {
      // ignore
    }
    unlockedParts.push(
      `${validAvatars.length} Character Avatar${validAvatars.length === 1 ? '' : 's'}`
    );
    bonusFandomPts += validAvatars.length * 10;
  }

  if (creatorsToUnlock.length > 0) {

  profilePatch.fandom_score = (Number(currentProf?.fandom_score) || 0) + bonusFandomPts;

  const { error: profErr } = await supabase
    .from('profiles')

export const redeemUniversalCode = async (

    throw new Error('Please enter a valid code or PIN.');

  if (cleanCode.startsWith('SAM-CR-')) {
    localStorage.setItem('am_pending_creator_invite', cleanCode);

      status: 'creator_invite',

      title: 'Creator Invite Detected',
      message: 'Redirecting to Creator Onboarding...'

        .from('profiles')
        .select('is_premium, referral_commission_paid_at, has_used_referral_trial')
        .eq('id', activeUserId)
        .maybeSingle();

      if (pCheck && !pCheck.is_premium && !pCheck.referral_commission_paid_at) {
        await supabase
          .from('profiles')
          .update({
            referred_by_code: cleanCode,
            referred_at: new Date().toISOString()
          })
          .eq('id', activeUserId);
      }
    }
    return {
      status: 'referral_code',
      code: cleanCode,
      title: alreadyUsedTrial ? 'Partner Code Applied!' : '3-Day Free Trial Unlocked!',
      message: alreadyUsedTrial
        ? `Partner Referral Code "${cleanCode}" applied to your $20/yr Annual Pass!`
        : `Partner Referral Code "${cleanCode}" applied! Your 3-Day Free Trial on the $20/yr Annual Pass is now unlocked.`
    };
  }

  const { data: currentProf } = await supabase
    .from('profiles')
    .select('*')
    .eq('id', activeUserId)
    .maybeSingle();

  const alreadyPremium = Boolean(isAlreadyPremiumHint || currentProf?.is_premium);

  // 3. 5-Digit Creator Autograph PIN
  if (/^\d{5}$/.test(cleanCode)) {
    const { data: pinData } = await supabase
      .from('bingo_codes')
      .select('*')
      .eq('code', cleanCode)
      .maybeSingle();

    if (!pinData) {
      throw new Error('Invalid 5-digit Autograph PIN. Please check the number and try again.');

        pinMax === 1
          ? 'This 1-Time Fan Gift PIN has already been claimed!'
          : 'This PIN has reached its maximum redemptions.'
      );
    }

    await supabase
      .from('bingo_codes')
      .update({ times_used: pinUsed + 1 })
      .eq('id', pinData.id);

    const unlockedCreator = await unlockCreatorAutographMaster(
      activeUserId,
      pinData.creator_name,
      pinData.event_name || 'Live Event PIN Unlock'
    );

    const existingAutographs: string[] = Array.isArray(currentProf?.unlocked_autographs)
      ? currentProf.unlocked_autographs.map(String)
      : [];
    const nextAutographs = Array.from(
      new Set([...existingAutographs, pinData.creator_name, unlockedCreator].filter(Boolean))
    );

    const pinPatch: Record<string, any> = {
      fandom_score: (Number(currentProf?.fandom_score) || 0) + 25,
      unlocked_autographs: nextAutographs
    };

    const { error: pinProfErr } = await supabase
      .from('profiles')
      .update(pinPatch)
      .eq('id', activeUserId);

    if (pinProfErr) {
      await supabase
        .from('profiles')
        .update({ fandom_score: (Number(currentProf?.fandom_score) || 0) + 25 })
        .eq('id', activeUserId);
    }

    window.dispatchEvent(new Event('profileUpdated'));
    window.dispatchEvent(new Event('bingoUpdated'));

    return {
      status: 'success',
      code: cleanCode,
      title: 'Autograph Unlocked!',
      message: `You unlocked ${
        unlockedCreator || pinData.creator_name
      }'s Gold Autograph (+25 Fandom Pts)! It is now displayed in your Bingo Book and on their Series pages.`,
      unlockedCreator: unlockedCreator || pinData.creator_name
    };
  }

  // 4. Custom Promo / Flash Bundle / Gift Sub Code from `promo_codes`
  const { data: promoRow } = await supabase
    .from('promo_codes')
    .select('*')
    .ilike('code', cleanCode)
    .maybeSingle();

  if (promoRow) {
    const now = new Date();
    if (promoRow.expires_at && new Date(promoRow.expires_at) < now) {
      throw new Error('This bundle or access code has expired.');
    }
    const timesUsed = Number(promoRow.times_used || 0);
    const maxUses = Number(promoRow.max_uses || 1);
    if (timesUsed >= maxUses) {
      throw new Error('All available claims for this code have already been redeemed!');
    }

    const rawBundle: any =
      typeof promoRow.bundle_config === 'object' && promoRow.bundle_config !== null
        ? promoRow.bundle_config
        : {};

    const isCreatorMonthlyGift = Boolean(
      rawBundle.is_creator_monthly_gift ||
        rawBundle.non_stackable ||
        promoRow.gift_month_key ||
        cleanCode.startsWith('GIFT-')
    );

    const hasCustomRewards =
      !isCreatorMonthlyGift &&
      Boolean(
        promoRow.creator_name ||
          rawBundle.creator_autograph ||
          rawBundle.creator_autograph_2 ||
          rawBundle.card_skin_id ||
          rawBundle.card_skin_url ||
          rawBundle.card_skin_2_id ||
          rawBundle.card_skin_2_url ||
          (rawBundle.avatar_urls && rawBundle.avatar_urls.length > 0) ||
          Number(rawBundle.bonus_hypes || 0) > 0
      );

    if (alreadyPremium && (isCreatorMonthlyGift || !hasCustomRewards)) {
      throw new Error(
        'You already have an active Saturday AM+ PLUS membership! Your code was not used so you can save it or gift it to a friend.'
      );
    }

    const { error: updatePromoErr } = await supabase
      .from('promo_codes')
      .update({ times_used: timesUsed + 1 })
      .eq('id', promoRow.id);
    if (updatePromoErr) throw updatePromoErr;

    const { unlockedParts, unlockedCreators, activatedPlus } =
      await grantMasterBundleRewardsToUser(
        activeUserId,
        {
          include_1yr_sub: isCreatorMonthlyGift ? true : rawBundle.include_1yr_sub !== false,
          card_skin_id: rawBundle.card_skin_id,
          card_skin_name: rawBundle.card_skin_name,
          card_skin_url: rawBundle.card_skin_url,
          card_skin_2_id: rawBundle.card_skin_2_id,
          card_skin_2_name: rawBundle.card_skin_2_name,
          card_skin_2_url: rawBundle.card_skin_2_url,
          avatar_urls: rawBundle.avatar_urls || [],
          creator_autograph: rawBundle.creator_autograph || promoRow.creator_name,
          creator_autograph_2: rawBundle.creator_autograph_2,
          bonus_hypes: rawBundle.bonus_hypes || 0,
          bundle_title:
            rawBundle.bundle_title ||
            (isCreatorMonthlyGift
              ? `${promoRow.creator_name || 'Creator'} 1-Year PLUS Gift Sub`
              : undefined)
        },
        alreadyPremium,
        rawBundle.bundle_title || `Code: ${cleanCode}`
      );

    return {
      status: 'success',
      code: cleanCode,
      title: rawBundle.bundle_title
        ? `${rawBundle.bundle_title} Unlocked!`
        : unlockedCreators[0]
        ? 'Flash Bundle Unlocked!'
        : '1-Year PLUS Activated!',
      message: `Success! You unlocked: ${unlockedParts.join(' • ')}!`,
      unlockedCreator: unlockedCreators[0],
      unlockedParts,
      activatedPlus
    };
  }

  // 5. Standard 1-Year Store Batch Access Key
  if (alreadyPremium) {
    throw new Error(
      'You already have an active Saturday AM+ PLUS membership! Your store code was not used so you can save it or gift it to a friend.'
    );
  }

  const { data: rpcSuccess, error: rpcErr } = await supabase.rpc('redeem_premium_code', {
    input_code: cleanCode,
    user_uuid: activeUserId
  });

  if (rpcErr || !rpcSuccess) {
    throw new Error('Invalid or already redeemed code or PIN.');
  }

  const curActive = Math.max(
    0,
    Number(currentProf?.hypes_remaining ?? currentProf?.super_hypes_left ?? 1)
  );
  const nextActive = Math.min(10, Math.max(3, curActive + 2));

  await supabase
    .from('profiles')
    .update({
      is_premium: true,
      hypes_remaining: nextActive,
      super_hypes_left: nextActive
    })
    .eq('id', activeUserId);

  window.dispatchEvent(new Event('profileUpdated'));

  return {
    status: 'success',
    code: cleanCode,
    title: 'PLUS Membership Activated!',
    message: 'Success! Your 1-Year Saturday AM+ PLUS membership is now active!',
    activatedPlus: true
  };
};

const isNewRelease = (publishAt?: string | null, createdAt?: string | null) => {
  const now = Date.now();
  const threeDaysMs = 3 * 24 * 60 * 60 * 1000;
  const earlyAccessBufferMs = 24 * 60 * 60 * 1000;

  const candidates = [publishAt, createdAt].filter(Boolean);
  for (const dateStr of candidates) {
    const t = new Date(dateStr as string).getTime();
    if (!isNaN(t) && t <= now + earlyAccessBufferMs && now - t < threeDaysMs) {
      return true;
    }
  }
  return false;
};

const isWithin48HourDebutSpotlight = (seriesData: any, latestActivityMs: number, allSeries: any[]) => {
  if (!seriesData) return false;
  const now = Date.now();
  const fortyEightHoursMs = 48 * 60 * 60 * 1000;
  const earlyBufferMs = 24 * 60 * 60 * 1000;

  const seriesReleaseMs = new Date(seriesData.publish_at || seriesData.created_at || 0).getTime();
  const effectiveDebutMs = !isNaN(seriesReleaseMs) && seriesReleaseMs > 0 ? seriesReleaseMs : latestActivityMs;

  if (effectiveDebutMs <= 0) return false;
  const ageMs = now - effectiveDebutMs;
  if (ageMs < -earlyBufferMs || ageMs > fortyEightHoursMs) return false;

  return isInOneShotDebutWindow(seriesData, allSeries) || ageMs <= fortyEightHoursMs;
};

};

          isActiveOneShotWednesday(seriesData, seriesList) &&
          getOneShotRemainingMs(seriesData, seriesList) > 0
      );

      const chTime = new Date(ch.publish_at || ch.created_at || 0).getTime();
      const validTime = isNaN(chTime) ? 0 : chTime;
      const existing = highestChapterBySeries.get(slug);

      if (!existing) {
        highestChapterBySeries.set(slug, {
          chapter: ch,
          latestSeriesActivityMs: validTime,
          isActiveOsw
        });
      } else {
        const bestActivityMs = Math.max(existing.latestSeriesActivityMs, validTime);
        const shouldReplaceChapter = isActiveOsw
          ? Number(ch.chapter_number) < Number(existing.chapter.chapter_number)
          : Number(ch.chapter_number) > Number(existing.chapter.chapter_number);

        highestChapterBySeries.set(slug, {
          chapter: shouldReplaceChapter ? ch : existing.chapter,
          latestSeriesActivityMs: bestActivityMs,
          isActiveOsw: existing.isActiveOsw || isActiveOsw
        });
      }
    });

    seriesList.forEach((s: any) => {
      if (!s?.slug || s.is_hidden) return;
      const slug = String(s.slug);
      if (highestChapterBySeries.has(slug)) return;
      const isActiveOsw =
        isActiveOneShotWednesday(s, seriesList) &&
        getOneShotRemainingMs(s, seriesList) > 0;
      if (isActiveOsw) {
        const sTime = new Date(s.publish_at || s.created_at || 0).getTime();
        highestChapterBySeries.set(slug, {
          chapter: {
            id: `osw_${slug}`,
            series_slug: slug,
            chapter_number: 1,
            title: s.title || 'One-Shot Wednesday',
            thumbnail_url: s.cover_url || s.character_url || '',
            is_published: true,
            publish_at: s.publish_at || s.created_at
          },
          latestSeriesActivityMs: isNaN(sTime) ? 0 : sTime,
          isActiveOsw: true
        });
      }
    });

    const allSeriesEntries = Array.from(highestChapterBySeries.values());

    const freshEntries = allSeriesEntries.filter(
      entry => entry.isActiveOsw || now - entry.latestSeriesActivityMs <= maxLatestAgeMs
    );
    const candidatePool = freshEntries.length >= 6 ? freshEntries : allSeriesEntries;

    return candidatePool
      .sort((a, b) => {
        if (a.isActiveOsw && !b.isActiveOsw) return -1;
        if (!a.isActiveOsw && b.isActiveOsw) return 1;

        const aRead = readChapterIdSet.has(String(a.chapter.id));
        const bRead = readChapterIdSet.has(String(b.chapter.id));

        if (aRead !== bRead) return aRead ? 1 : -1;

        const seriesA = seriesList.find((s: any) => s.slug === a.chapter.series_slug);
        const seriesB = seriesList.find((s: any) => s.slug === b.chapter.series_slug);
        const aIsFreshDebut = isWithin48HourDebutSpotlight(seriesA, a.latestSeriesActivityMs, seriesList);
        const bIsFreshDebut = isWithin48HourDebutSpotlight(seriesB, b.latestSeriesActivityMs, seriesList);
        if (aIsFreshDebut && !bIsFreshDebut) return -1;
        if (!aIsFreshDebut && bIsFreshDebut) return 1;

        return b.latestSeriesActivityMs - a.latestSeriesActivityMs;
      })
      .map(entry => ({
        ...entry.chapter,
        isActiveOsw: entry.isActiveOsw,
        isAlreadyRead: readChapterIdSet.has(String(entry.chapter.id)),
        publish_at:
          entry.latestSeriesActivityMs > 0
            ? new Date(entry.latestSeriesActivityMs).toISOString()
            : entry.chapter.publish_at
      }))
      .slice(0, 8);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [latestChapters, seriesList, nowTick, secondTick, isPlusSubscriber, readChapterIdSet]);

  const visibleNotifications = useMemo(() => {
    const chaptersById = new Map<string, any>();
    latestChapters.forEach((ch: any) => chaptersById.set(String(ch.id), ch));

    const unifiedMap = new Map<string, any>();
    const isDismissed = (key: string) => dismissedNotifs.includes(key);

    notifications.forEach((n: any) => {
      if (!n || !n.id || !n.title) return;
      const sourceKey = String(n.source_key || '');
      if (sourceKey.startsWith('chapter_')) {
        const chapId = sourceKey.replace('chapter_', '');
        const matchedChap = chaptersById.get(chapId);
        if (!matchedChap || !matchedChap.is_published) return;
        const parentSeries = seriesList.find((s: any) => s.slug === matchedChap.series_slug);
        if (!parentSeries || parentSeries.is_hidden) return;
        const effectivePublishAt = matchedChap.publish_at || n.publish_at;
        const isFreeOswOrDebut = isActiveOneShotWednesday(parentSeries, seriesList);
        if (!isFreeOswOrDebut && !isChapterReleasedForUser(effectivePublishAt, isPlusSubscriber)) return;
        const dedupeKey = `chapter_${chapId}`;
        if (isDismissed(String(n.id)) || isDismissed(dedupeKey)) return;
        unifiedMap.set(dedupeKey, {
          ...n, id: String(n.id), dedupeKey, chapter_id: matchedChap.id, link_target: parentSeries.slug,
          thumbnail_url: matchedChap.thumbnail_url || parentSeries.cover_url || n.thumbnail_url,
          publish_at: effectivePublishAt, created_at: matchedChap.created_at || n.created_at
        });
      } else if (sourceKey.startsWith('series_')) {
        const sSlug = sourceKey.replace('series_', '');
        const matchedSeries = seriesList.find((s: any) => s.slug === sSlug);
        if (!matchedSeries || matchedSeries.is_hidden) return;
        const effectivePublishAt = matchedSeries.publish_at || n.publish_at;
        const isFreeOswOrDebut = isActiveOneShotWednesday(matchedSeries, seriesList);
        if (!isFreeOswOrDebut && !isReleasedYet(effectivePublishAt)) return;
        const dedupeKey = `series_${sSlug}`;
        if (isDismissed(String(n.id)) || isDismissed(dedupeKey)) return;
        unifiedMap.set(dedupeKey, {
          ...n, id: String(n.id), dedupeKey, link_target: matchedSeries.slug,
          thumbnail_url: matchedSeries.cover_url || matchedSeries.character_url || n.thumbnail_url,
          publish_at: effectivePublishAt, created_at: matchedSeries.created_at || n.created_at
        });
      } else {
        if (isDismissed(String(n.id))) return;
        if (!isChapterReleasedForUser(n.publish_at, isPlusSubscriber)) return;
        if (n.link_target && !n.link_target.startsWith('http') && n.link_target !== 'news') {
          const matchedSeries = seriesList.find((s: any) => s.slug === n.link_target);
          if (matchedSeries && (matchedSeries.is_hidden || !isReleasedYet(matchedSeries.publish_at))) return;
        }
        unifiedMap.set(`custom_${n.id}`, { ...n, id: String(n.id), dedupeKey: `custom_${n.id}` });
      }
    });

    visibleLatestChapters.forEach((ch: any) => {
      const dedupeKey = `chapter_${ch.id}`;
      if (unifiedMap.has(dedupeKey) || isDismissed(dedupeKey) || isDismissed(String(ch.id))) return;
      const parentSeries = seriesList.find((s: any) => s.slug === ch.series_slug);
      if (!parentSeries) return;
      const alreadyHasManualMatch = Array.from(unifiedMap.values()).some(
        existing => existing.link_target === parentSeries.slug && String(existing.message || '').includes(`Chapter ${ch.chapter_number}`)
      );
      if (alreadyHasManualMatch) return;
      unifiedMap.set(dedupeKey, {
        id: dedupeKey, dedupeKey, chapter_id: ch.id, title: 'New Chapter Drop!',
        message: `${parentSeries.title} • Chapter ${ch.chapter_number}${ch.title ? `: "${ch.title}"` : ''} is now live!`,
        thumbnail_url: ch.thumbnail_url || parentSeries.cover_url || null,
        link_target: parentSeries.slug, publish_at: ch.publish_at || ch.created_at, created_at: ch.created_at
      });
    });

    seriesList.forEach((s: any) => {
      if (!s?.slug || s.is_hidden) return;
      const effectiveSeriesDate = s.publish_at || s.created_at;
      if (!isRecentAddition(effectiveSeriesDate, 14)) return;
      const dedupeKey = `series_${s.slug}`;
      if (unifiedMap.has(dedupeKey) || isDismissed(dedupeKey)) return;
      const hasCustomSeriesNotif = Array.from(unifiedMap.values()).some(
        existing => existing.link_target === s.slug && String(existing.title || '').toLowerCase().includes('series')
      );
      if (hasCustomSeriesNotif) return;
      unifiedMap.set(dedupeKey, {
        id: dedupeKey, dedupeKey, title: 'New Series Drop!',
        message: `${s.title}${s.creator_name ? ` by ${s.creator_name}` : ''} is now on Saturday AM! Tap to start reading.`,
        thumbnail_url: s.cover_url || s.character_url || null, link_target: s.slug,
        publish_at: effectiveSeriesDate, created_at: s.created_at || effectiveSeriesDate
      });
    });

    const seenCreatorNames = new Set<string>();
    recentCreators.forEach((cr: any) => {
      if (!cr?.name || cr.is_visible === false) return;
      const cleanName = String(cr.name).trim();
      const normName = cleanName.toLowerCase();
      if (seenCreatorNames.has(normName)) return;
      seenCreatorNames.add(normName);
      if (!isRecentAddition(cr.created_at, 14)) return;
      const dedupeKey = `creator_${normName.replace(/\s+/g, '_')}`;
      if (unifiedMap.has(dedupeKey) || isDismissed(dedupeKey)) return;
      const parentSeries = seriesList.find((s: any) => s.slug === cr.series_slug);
      unifiedMap.set(dedupeKey, {
        id: dedupeKey, dedupeKey, title: 'Meet the Creator!',
        message: `${cleanName}${parentSeries?.title ? ` (${parentSeries.title})` : ''} has been added to the Creators Roster. Tap to view their bio & works!`,
        thumbnail_url: cr.avatar_url || parentSeries?.creator_avatar || parentSeries?.cover_url || `${CLOUDFLARE_BASE_URL}/assets/creator-avatar.jpg`,
        navPayload: { action: 'creators', targetCreatorName: cleanName }, publish_at: cr.created_at, created_at: cr.created_at
      });
    });

    recentCharacters.forEach((ch: any) => {
      if (!ch?.id || !ch?.name || ch.is_hidden) return;
      if (!isRecentAddition(ch.created_at, 14)) return;
      const dedupeKey = `character_${ch.id}`;
      if (unifiedMap.has(dedupeKey) || isDismissed(dedupeKey)) return;
      const parentSeries = seriesList.find((s: any) => s.slug === ch.series_slug);
      unifiedMap.set(dedupeKey, {
        id: dedupeKey, dedupeKey, title: 'New Character Unlocked!',
        message: `${ch.name}${parentSeries?.title ? ` from ${parentSeries.title}` : ''} joined the Character Roster! Tap to inspect their combat file.`,
        thumbnail_url: ch.headshot_url || parentSeries?.cover_url || null,
        navPayload: { action: 'characters', targetCharacterId: ch.id, targetCharacterName: ch.name },
        publish_at: ch.created_at, created_at: ch.created_at
      });
    });

    (cardSkins || []).forEach((skin: any) => {
      if (!skin?.id || skin.is_active === false || !isRecentAddition(skin.created_at, 14)) return;
      const dedupeKey = `skin_${skin.id}`;
      if (unifiedMap.has(dedupeKey) || isDismissed(dedupeKey)) return;
      unifiedMap.set(dedupeKey, {
        id: dedupeKey, dedupeKey, title: 'New Hype Card Skin!',
        message: `The "${skin.name}" Hype Card skin is now available in the Vault! Tap to equip it to your pass.`,
        thumbnail_url: skin.image_url || null, navPayload: { action: 'profile', openLoadoutTab: 'card' },
        publish_at: skin.created_at, created_at: skin.created_at
      });
    });

    (vaultAvatars || []).forEach((av: any) => {
      if (!av?.id || !isRecentAddition(av.created_at, 14)) return;
      const dedupeKey = `avatar_${av.id}`;
      if (unifiedMap.has(dedupeKey) || isDismissed(dedupeKey)) return;
      unifiedMap.set(dedupeKey, {
        id: dedupeKey, dedupeKey, title: 'New Avatar Art Added!',
        message: `New "${av.name || 'Character'}" avatar art is live! Tap to customize your profile avatar.`,
        thumbnail_url: av.image_url || null, navPayload: { action: 'profile', openLoadoutTab: 'art' },
        publish_at: av.created_at, created_at: av.created_at
      });
    });

    (vaultFrames || []).forEach((fr: any) => {
      if (!fr?.id || fr.is_active === false || !isRecentAddition(fr.created_at, 14)) return;
      const dedupeKey = `frame_${fr.id}`;
      if (unifiedMap.has(dedupeKey) || isDismissed(dedupeKey)) return;
      unifiedMap.set(dedupeKey, {
        id: dedupeKey, dedupeKey, title: 'New Border Frame Added!',
        message: `The "${fr.name}" (${fr.tier || 'Basic'}) border frame has been added! Tap to preview & equip it.`,
        thumbnail_url: `${CLOUDFLARE_BASE_URL}/other%20icons/Hype%20flame.png`,
        navPayload: { action: 'profile', openLoadoutTab: 'frame' },
        publish_at: fr.created_at, created_at: fr.created_at
      });
    });

    return Array.from(unifiedMap.values())
      .sort((a, b) => new Date(b.publish_at || b.created_at || 0).getTime() - new Date(a.publish_at || a.created_at || 0).getTime())
      .slice(0, 20);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [notifications, visibleLatestChapters, latestChapters, recentCreators, recentCharacters, cardSkins, vaultAvatars, vaultFrames, seriesList, dismissedNotifs, nowTick, isPlusSubscriber]);

  const fetchGlobalNotifications = async () => {
    try {
      const { data: notifData } = await supabase.from('app_notifications').select('*').order('created_at', { ascending: false }).limit(40);
      if (notifData) {
        const validData = notifData.filter(n => n && n.id && n.title);
        setNotifications(validData);
        memNotifications = validData;
      }
    } catch (e) { console.error('Error fetching notifs:', e); }
  };

  useEffect(() => {
    const savedDismissed = JSON.parse(localStorage.getItem('am_dismissed_notifs') || '[]');
    setDismissedNotifs(savedDismissed);

    const fetchHomeData = async (force = false) => {
      if (!force && memHeroSlides && memHomeSections && memLatestChapters && Date.now() - lastHomeFetchTime < 3 * 60 * 1000) {
        setHeroSlides(memHeroSlides); setHomeSections(memHomeSections); setLatestChapters(memLatestChapters);
        if (memRecentCreators) setRecentCreators(memRecentCreators);
        if (memRecentCharacters) setRecentCharacters(memRecentCharacters);
        setIsLoadingSlides(false);
        return;
      }

      try {
        const [
          { data: slideData },
          { data: sectionData },
          { data: chapData },
          { data: creatorsData },
          { data: charsData }
        ] = await Promise.all([
          supabase.from('hero_slides').select('id, desktop_url, mobile_url, link_type, link_target').order('id', { ascending: true }),
          supabase.from('home_sections').select('title, display_order').order('display_order', { ascending: true }),
          supabase
            .from('chapters')
            .select('id, series_slug, chapter_number, title, thumbnail_url, is_published, publish_at, created_at')
            .eq('is_published', true)
            .order('chapter_number', { ascending: true })
            .limit(2000),
          supabase.from('series_creators').select('id, name, series_slug, avatar_url, is_visible, created_at').order('id', { ascending: false }).limit(30),
          supabase.from('series_characters').select('id, name, series_slug, headshot_url, is_hidden, created_at').order('id', { ascending: false }).limit(30),
          fetchGlobalNotifications()
        ]);

        if (slideData) { setHeroSlides(slideData); memHeroSlides = slideData; }
        if (sectionData) { setHomeSections(sectionData); memHomeSections = sectionData; }
        if (chapData) { setLatestChapters(chapData); memLatestChapters = chapData; }
        if (creatorsData) { setRecentCreators(creatorsData); memRecentCreators = creatorsData; }
        if (charsData) { setRecentCharacters(charsData); memRecentCharacters = charsData; }
        lastHomeFetchTime = Date.now();
      } catch (err: any) { console.error('Error fetching home data:', err.message); }
      finally { setIsLoadingSlides(false); }
    };

    fetchHomeData();

    const handleSeriesOrChaptersUpdated = () => fetchHomeData(true);
    const handleVisibilityRefresh = () => {
      if (document.visibilityState === 'visible' && Date.now() - lastHomeFetchTime > 5 * 60 * 1000) fetchHomeData(true);
    };

    window.addEventListener('seriesUpdated', handleSeriesOrChaptersUpdated);
    window.addEventListener('vaultUpdated', handleSeriesOrChaptersUpdated);
    document.addEventListener('visibilitychange', handleVisibilityRefresh);
    return () => {
      window.removeEventListener('seriesUpdated', handleSeriesOrChaptersUpdated);
      window.removeEventListener('vaultUpdated', handleSeriesOrChaptersUpdated);
      document.removeEventListener('visibilitychange', handleVisibilityRefresh);
    };
  }, []);

  useEffect(() => { if (showNotifications) fetchGlobalNotifications(); }, [showNotifications]);

  const handleDismissNotif = (e: React.MouseEvent, notif: any) => {
    e.stopPropagation();
    const keysToDismiss = [String(notif.id), String(notif.dedupeKey || '')].filter(Boolean);
    const updated = Array.from(new Set([...dismissedNotifs, ...keysToDismiss]));
    setDismissedNotifs(updated);
    localStorage.setItem('am_dismissed_notifs', JSON.stringify(updated));
  };

  const handleClearAllNotifs = (e: React.MouseEvent) => {
    e.stopPropagation();
    const allLoadedIds = visibleNotifications.flatMap(n => [String(n.id), String(n.dedupeKey || '')].filter(Boolean));
    const updated = Array.from(new Set([...dismissedNotifs, ...allLoadedIds]));
    setDismissedNotifs(updated);
    localStorage.setItem('am_dismissed_notifs', JSON.stringify(updated));
  };

  const handleNotificationClick = (n: any) => {
    setShowNotifications(false);
    if (n.navPayload) {
      if (n.navPayload.targetCreatorName) sessionStorage.setItem('am_target_creator', n.navPayload.targetCreatorName);
      if (n.navPayload.targetCharacterId) sessionStorage.setItem('am_target_character_id', String(n.navPayload.targetCharacterId));
      if (n.navPayload.targetCharacterName) sessionStorage.setItem('am_target_character_name', String(n.navPayload.targetCharacterName));
      if (n.navPayload.openLoadoutTab) sessionStorage.setItem('am_open_loadout_tab', String(n.navPayload.openLoadoutTab));
      onNavigate(n.navPayload);
      return;
    }
    if (n.link_target) {
      const target = String(n.link_target).trim();
      if (target.startsWith('http')) { window.open(target, '_blank'); return; }
      const matchedSeries = seriesList.find((s: any) => s.slug === target);
      if (matchedSeries && !matchedSeries.is_hidden) {
        onNavigate(n.chapter_id ? { ...matchedSeries, autoOpenChapterId: n.chapter_id } : matchedSeries);
      } else onNavigate({ action: target });
    }
  };

  useEffect(() => {
    const fetchRecentReads = async (forceRefresh = false) => {
      if (isLoading || !currentUser) {
        setRecentReads([]);
        setCompletedChapterIds(new Set());
        return;
      }
      if (!forceRefresh && memRecentReads && memCompletedChapterIds) {
        setRecentReads(memRecentReads);
        setCompletedChapterIds(memCompletedChapterIds);
        return;
      }
      try {
        const [{ data: history }, { data: completedRows }] = await Promise.all([
          supabase
            .from('reading_history')
            .select('*')
            .eq('user_id', currentUser.id)
            .order('updated_at', { ascending: false })
            .limit(25),
          supabase
            .from('completed_reads')
            .select('chapter_id')
            .eq('user_id', currentUser.id)
            .limit(500)
        ]);

        const compSet = new Set<string>(
          (completedRows || []).map((r: any) => String(r.chapter_id))
        );
        setCompletedChapterIds(compSet);
        memCompletedChapterIds = compSet;

        if (!history || history.length === 0) {
          setRecentReads([]);
          memRecentReads = [];
          return;
        }
        const allIds = history.map((h: any) => h.chapter_id);
        let chapters: any[] = [];
        if (allIds.length > 0) {
          const { data: chapterData } = await supabase.from('chapters').select('*').in('id', allIds);
          if (chapterData) chapters = chapterData;
        }

        // Since `history` is ordered by `updated_at DESC`, the first entry we see for each `series_slug`
        // is the latest chapter and page the user was reading in that series.
        const seenSeriesSlugs = new Set<string>();
        const combined: any[] = [];

        for (const h of history) {
          const chapIdStr = String(h.chapter_id);
          const chap = chapters.find((c: any) => String(c.id) === chapIdStr);
          if (!chap || !chap.is_published) continue;

          const slug = String(chap.series_slug || '').trim().toLowerCase();
          if (!slug || seenSeriesSlugs.has(slug)) continue;

          const series = seriesList.find((s: any) => String(s.slug || '').toLowerCase() === slug);
          if (!series || series.is_hidden) continue;

          const isFreeOswOrDebut = isActiveOneShotWednesday(series, seriesList);
          if (!isFreeOswOrDebut && !isChapterReleasedForUser(chap.publish_at, isPlusSubscriber)) {
            continue;
          }

          // Mark this series as seen so older chapters of the same series never repeat in Jump Back In
          seenSeriesSlugs.add(slug);

          // If the user's most recent chapter in this series was completed, hide the series from Jump Back In
          if (compSet.has(chapIdStr)) continue;
          const totalPages = Number(chap.page_count || 0);
          if (totalPages > 1 && Number(h.page_index || 0) >= totalPages - 1) {
            continue;
          }

          combined.push({
            ...h,
            type: 'series',
            target: { ...series, action: 'series' },
            title: series.title,
            subtitle: `Chapter ${chap.chapter_number}`,
            image: chap.thumbnail_url || series.cover_url
          });
        }

        setRecentReads(combined);
        memRecentReads = combined;
      } catch (err: any) {
        console.error('Critical Error fetching recent reads:', err);
      }
    };

    fetchRecentReads();

    const handleProgressOrProfileRefresh = () => {
      memRecentReads = null;
      memCompletedChapterIds = null;
      fetchRecentReads(true);
    };

    window.addEventListener('progressUpdated', handleProgressOrProfileRefresh);
    window.addEventListener('profileUpdated', handleProgressOrProfileRefresh);
    return () => {
      window.removeEventListener('progressUpdated', handleProgressOrProfileRefresh);
      window.removeEventListener('profileUpdated', handleProgressOrProfileRefresh);
    };
  }, [seriesList, isLoading, currentUser, isPlusSubscriber]);

  useEffect(() => {
    const timer = setInterval(() => setCurrentSlide(prev => (heroSlides.length > 0 ? (prev + 1) % heroSlides.length : 0)), 9000);
    return () => clearInterval(timer);
  }, [heroSlides.length]);

  useEffect(() => {
    const handleInstantLogout = () => {
      memRecentReads = null;
      memCompletedChapterIds = null;
      setRecentReads([]);
      setCompletedChapterIds(new Set());
    };
    window.addEventListener('instantLogout', handleInstantLogout);
    return () => window.removeEventListener('instantLogout', handleInstantLogout);
  }, []);

  const handleSlideClick = (slide: any) => {
    trackEvent('banner_click', { target: slide.link_target, link_type: slide.link_type });
    if (!slide.link_target) return;
    if (slide.link_type === 'external') window.open(slide.link_target, '_blank', 'noopener,noreferrer');
    else if (slide.link_type === 'series') {
      const matchedSeries = seriesList.find((s: any) => s.slug === slide.link_target);
      if (matchedSeries && !matchedSeries.is_hidden) onNavigate(matchedSeries);
    }
  };

  const nextScheduledBySeries = useMemo(() => {
    const map: Record<string, string> = {};
    latestChapters.forEach((ch: any) => {
      if (!ch.is_published || !ch.publish_at) return;
      if (!isChapterReleasedForUser(ch.publish_at, isPlusSubscriber)) {
        const existing = map[ch.series_slug];
        if (!existing || new Date(ch.publish_at).getTime() < new Date(existing).getTime()) {
          map[ch.series_slug] = ch.publish_at;
        }
      }
    });
    return map;
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [latestChapters, isPlusSubscriber, nowTick]);

  if (isLoading || isLoadingSlides) {
    return (
      <div className="min-h-screen bg-transparent flex flex-col items-center justify-center gap-4 pb-20">
        <div className="relative flex items-center justify-center animate-hype-glow">
          <img
            src="https://pub-180171f859f64aa7aadb7001a6b96e65.r2.dev/other%20icons/Hype%20flame.png"
            alt="Saturday AM Loading"
            className="w-24 h-24 sm:w-28 sm:h-28 object-contain"
            style={{ filter: 'url(#hype-flame-color)' }}
          />
        </div>
      </div>
    );
  }

  return (
    <div className="relative min-h-screen bg-transparent text-white pb-24">
      <FeatureTutorialModal tutorialId="home_page_basics" slides={homeTutorialSlides} forceOpen={forceTutorial} onClose={() => setForceTutorial(false)} />

                    <h3 className="text-white font-black uppercase text-xs md:text-sm leading-tight line-clamp-1">{read.title}</h3>
                    <div className="flex justify-between items-center mt-1">
                      <p className="text-[#fe9a00] font-bold text-[9px] md:text-[10px] uppercase tracking-widest">{read.subtitle}</p>
                      <p className="text-zinc-300 font-bold text-[8px] md:text-[9px] uppercase tracking-widest bg-black/60 px-1.5 py-0.5 rounded border border-zinc-700">Pg. {read.page_index + 1}</p>
                    </div>
                  </div>
                </div>
              ))}
            </div>
          </div>
        )}

        {(() => {
          // Detect if today in Eastern Time (America/New_York) is Monday (1), Wednesday (3), or Saturday (6)
          const getEstCalendarInfo = (d: Date) => {
            try {
              const parts = new Intl.DateTimeFormat('en-US', {
                timeZone: 'America/New_York',
                weekday: 'short',
                year: 'numeric',
                month: '2-digit',
                day: '2-digit'
              }).formatToParts(d);
              const wk = parts.find(p => p.type === 'weekday')?.value || '';
              const yr = parts.find(p => p.type === 'year')?.value || '';
              const mo = parts.find(p => p.type === 'month')?.value || '';
              const da = parts.find(p => p.type === 'day')?.value || '';
              const dowMap: Record<string, number> = {
                Sun: 0,
                Mon: 1,
                Tue: 2,
                Wed: 3,
                Thu: 4,
                Fri: 5,
                Sat: 6
              };
              return { dow: dowMap[wk] ?? d.getDay(), dateKey: `${yr}-${mo}-${da}` };
            } catch {
              return { dow: d.getDay(), dateKey: getLocalCalendarDateKey(d) };
            }
          };

          const todayEst = getEstCalendarInfo(new Date());
          const isOfficialReleaseDay =
            todayEst.dow === 1 || todayEst.dow === 3 || todayEst.dow === 6;

          const droppingTodayCounts = new Map<string, number>();
          if (isOfficialReleaseDay) {
            latestChapters.forEach((ch: any) => {
              if (!ch?.is_published || !ch?.series_slug || !ch?.publish_at) return;
              if (!isChapterReleasedForUser(ch.publish_at, isPlusSubscriber)) return;
              const chapEst = getEstCalendarInfo(new Date(ch.publish_at));
              if (chapEst.dateKey === todayEst.dateKey) {
                const slugKey = String(ch.series_slug).toLowerCase();
                droppingTodayCounts.set(slugKey, (droppingTodayCounts.get(slugKey) || 0) + 1);
              }
            });

            seriesList.forEach((s: any) => {
              if (!s?.slug || s.is_hidden) return;
              const slugKey = String(s.slug).toLowerCase();
              if (todayEst.dow === 3 && isActiveOneShotWednesday(s, seriesList)) {
                if (!droppingTodayCounts.has(slugKey)) droppingTodayCounts.set(slugKey, 1);
              } else if (isInOneShotDebutWindow(s, seriesList) && s.publish_at) {
                const sEst = getEstCalendarInfo(new Date(s.publish_at));
                if (sEst.dateKey === todayEst.dateKey && !droppingTodayCounts.has(slugKey)) {
                  droppingTodayCounts.set(slugKey, 1);
                }
              }
            });
          }

          return homeSections.map(section => {
            const seriesInSection = seriesList
              .filter(
                (s: any) =>
                  s.home_section === section.title &&
                  !s.is_hidden &&
                  isReleasedYet(s.publish_at)
              )
              .map((s: any) => {
                const dropCount = droppingTodayCounts.get(String(s.slug || '').toLowerCase()) || 0;
                return {
                  ...s,
                  isReleaseDayDrop: dropCount > 0,
                  releaseDayDropCount: dropCount
                };
              })
              .sort((a: any, b: any) => {
                if (a.isReleaseDayDrop && !b.isReleaseDayDrop) return -1;
                if (!a.isReleaseDayDrop && b.isReleaseDayDrop) return 1;
                return (a.display_order || 99) - (b.display_order || 99);
              });

            if (seriesInSection.length === 0) return null;
            return (
              <SeriesSection
                key={section.id}
                title={section.title}
                series={seriesInSection}
                onSeriesClick={onNavigate}
                currentUser={currentUser}
                userTier={userTier}
                nextScheduledBySeries={nextScheduledBySeries}
                onRequireAuth={onLoginClick}
              />
            );
          });
        })()}
      </div>
    </div>
  );
};
